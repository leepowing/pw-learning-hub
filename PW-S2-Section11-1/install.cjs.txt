// No dependencies. Run with Node.js from the extracted package directory.
const fs=require('node:fs'), path=require('node:path'), crypto=require('node:crypto');
const fail=message=>{throw new Error(message);};
const normal=s=>s.replace(/^\uFEFF/,'').replace(/\r\n/g,'\n');
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
function main(){
 if(!process.argv[2])fail('Usage: node install.cjs.txt PROJECT_PATH [--check]');
 const root=path.resolve(process.argv[2]);
 if(!fs.existsSync(path.join(root,'package.json')))fail('Project package.json not found.');
 const roots=['app','src/app'].filter(a=>fs.existsSync(path.join(root,a,'maths/s2/chapter-11/page.tsx')));
 if(roots.length!==1)fail('Expected one installed Chapter 11 homepage in app or src/app. Install the homepage first.');
 const app=roots[0], prefix=app==='app'?'':'src/';
 const home=path.join(root,app,'maths/s2/chapter-11/page.tsx');
 const expected=fs.readFileSync(path.join(__dirname,'payload/home-before.tsx.txt'),'utf8');
 let updated=expected;
 const replacements=[
  ['Chapter overview available · Lessons, Checkpoint and Flashcards coming soon','Chapter overview and Section 11.1 available · Remaining lessons, Checkpoint and Flashcards coming soon'],
  ['<span className="coming">Coming soon</span></article>)}</div></section>','{number===\'11.1\'?<Link href="/maths/s2/chapter-11/concept-of-nth-roots">Start lesson →</Link>:<span className="coming">Coming soon</span>}</article>)}</div></section>'],
 ];
 for(const [before,after] of replacements){if(updated.split(before).length!==2)fail('Invalid package homepage anchor.');updated=updated.replace(before,after);}
 const original=fs.readFileSync(home);
 if(normal(original.toString('utf8'))!==normal(expected)&&normal(original.toString('utf8'))!==normal(updated))fail('Homepage differs from the supplied version. No files changed. Please send your latest Chapter 11 page.tsx; do not overwrite custom content.');
 const jobs=[];
 if(normal(original.toString('utf8'))!==normal(updated)){
  const eol=original.toString('utf8').includes('\r\n')?'\r\n':'\n';
  const bom=original.toString('utf8').startsWith('\uFEFF')?'\uFEFF':'';
  jobs.push({target:home,before:original,data:Buffer.from(bom+normal(updated).replace(/\n/g,eol))});
 }
 for(const [source,relative] of [
  ['page.tsx.txt',`${app}/maths/s2/chapter-11/concept-of-nth-roots/page.tsx`],
  ['S2Chapter11Roots.tsx.txt',`${prefix}components/maths/S2Chapter11Roots.tsx`],
 ]){
  const target=path.join(root,relative),data=fs.readFileSync(path.join(__dirname,'payload',source));
  if(fs.existsSync(target)){if(!fs.readFileSync(target).equals(data))fail(`Conflict: ${relative} has different content. No files changed.`);}
  else jobs.push({target,before:null,data});
 }
 // Reject symlinks/junctions before any write, including the backup parent.
 for(const target of [home,...jobs.map(j=>j.target),path.join(root,'_pw-backups')]){
  let current=target;
  while(current===root||current.startsWith(root+path.sep)){
   if(fs.existsSync(current)&&fs.lstatSync(current).isSymbolicLink())fail('Symbolic link or junction encountered. No files changed.');
   if(current===root)break;current=path.dirname(current);
  }
 }
 if(!jobs.length){console.log('Already installed. No changes required.');return;}
 console.log('Planned files:\n'+jobs.map(j=>path.relative(root,j.target)).join('\n'));
 if(process.argv.includes('--check')){console.log('Preflight passed. No files changed.');return;}
 const backup=path.join(root,'_pw-backups','section11-1-'+Date.now()+'-'+crypto.randomBytes(3).toString('hex'));
 fs.mkdirSync(backup,{recursive:true});
 for(const job of jobs)if(job.before){const dest=path.join(backup,path.relative(root,job.target)+'.txt');fs.mkdirSync(path.dirname(dest),{recursive:true});fs.writeFileSync(dest,job.before,{flag:'wx'});}
 fs.writeFileSync(path.join(backup,'manifest.json'),JSON.stringify(jobs.map(j=>({path:path.relative(root,j.target),created:j.before===null,beforeSha256:j.before?sha(j.before):null,installedSha256:sha(j.data)})),null,2),{flag:'wx'});
 const done=[];
 try{
  for(const job of jobs){
   fs.mkdirSync(path.dirname(job.target),{recursive:true});
   if(job.before){
    if(!fs.readFileSync(job.target).equals(job.before))fail('File changed during installation.');
    done.push(job);fs.writeFileSync(job.target,job.data);
   }else{
    const fd=fs.openSync(job.target,'wx');done.push(job);
    try{fs.writeFileSync(fd,job.data);}finally{fs.closeSync(fd);}
   }
  }
 }catch(error){
  for(const job of done.reverse()){try{if(job.before)fs.writeFileSync(job.target,job.before);else fs.unlinkSync(job.target);}catch{console.error('Rollback incomplete for '+job.target+'; use backup '+backup);}}
  throw error;
 }
 console.log('Installed Section 11.1.\nBackup: '+backup+'\nNext: run npm run build in your project.');
}
try{main();}catch(error){console.error('STOP: '+error.message);process.exitCode=1;}
